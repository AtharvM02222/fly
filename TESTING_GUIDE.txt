TEST DATA GENERATION
--------------------

Generate test detections:

python << EOF
import requests
import random
from datetime import datetime, timedelta

api_key = "your-device-api-key"
base_lat, base_lon = 37.7749, -122.4194

for i in range(100):
    lat = base_lat + random.uniform(-0.01, 0.01)
    lon = base_lon + random.uniform(-0.01, 0.01)
    
    detection = {
        "detections": [{
            "client_detection_id": f"test-{i}",
            "latitude": lat,
            "longitude": lon,
            "severity": random.choice(["low", "medium", "high"]),
            "confidence": random.uniform(0.5, 0.95),
            "bbox": {
                "x1": random.uniform(0, 0.5),
                "y1": random.uniform(0, 0.5),
                "x2": random.uniform(0.5, 1.0),
                "y2": random.uniform(0.5, 1.0)
            },
            "model_version": "v1.0",
            "detected_at": (datetime.utcnow() - timedelta(hours=random.randint(0, 48))).isoformat() + "Z"
        }]
    }
    
    resp = requests.post(
        "http://localhost:8000/api/v1/detections",
        headers={"X-API-Key": api_key},
        json=detection
    )
    print(f"Detection {i}: {resp.status_code}")
EOF
